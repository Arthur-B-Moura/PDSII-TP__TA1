19 // Estima a distância em linha reta entre dois nós usando a fórmula de Haversine
20 // Serve como "chute" do A* — o custo mínimo possível entre dois pontos no mapa
21 double PathFinder::heuristic(long long from_id, long long to_id) const {
22  const MapNode& a = grafo_.get_node(from_id);
23  const MapNode& b = grafo_.get_node(to_id);
24 
25  double lat1 = deg2rad(a.posicao().latitude_);
26  double lat2 = deg2rad(b.posicao().latitude_);
27  double dlat = lat2 - lat1;
28  double dlon = deg2rad(b.posicao().longitude_ - a.posicao().longitude_);
29 
30  double x = sin(dlat / 2) * sin(dlat / 2) +
31  cos(lat1) * cos(lat2) *
32  sin(dlon / 2) * sin(dlon / 2);
33 
34  return 2.0 * R * atan2(sqrt(x), sqrt(1.0 - x));
35 }

38 // 0° = Norte, 90° = Leste, 180° = Sul, 270° = Oeste
39 double PathFinder::bearing(long long from_id, long long to_id) const {
40  const MapNode& a = grafo_.get_node(from_id);
41  const MapNode& b = grafo_.get_node(to_id);
42 
43  double lat1 = deg2rad(a.posicao().latitude_);
44  double lat2 = deg2rad(b.posicao().latitude_);
45  double dlon = deg2rad(b.posicao().longitude_ - a.posicao().longitude_);
46 
47  double y = sin(dlon) * cos(lat2);
48  double x = cos(lat1) * sin(lat2) - sin(lat1) * cos(lat2) * cos(dlon);
49 
50  double angle = atan2(y, x) * 180.0 / M_PI;
51  return fmod(angle + 360.0, 360.0);
52 }

72 // Algoritmo A*: encontra o caminho mais curto entre start_id e end_id
73 // Usa uma fila de prioridade para sempre explorar o nó mais promissor primeiro
74 // f(n) = g(n) + h(n), onde g = custo real acumulado, h = estimativa até o destino
75 PathResult PathFinder::find_path(long long start_id, long long end_id) const {
76  using pair_t = std::pair<double, long long>;
77 
78  // Fila de prioridade ordenada pelo menor f(n) — o nó mais promissor sai primeiro
79  std::priority_queue<pair_t, std::vector<pair_t>, std::greater<pair_t>> open_set;
80 
81  // g_score[n] = menor custo real conhecido do início até n
82  std::unordered_map<long long, double> g_score;
83 
84  // came_from[n] = nó anterior no caminho ótimo até n
85  std::unordered_map<long long, long long> came_from;
86 
87  g_score[start_id] = 0.0;
88  open_set.push({heuristic(start_id, end_id), start_id});
89 
90  while (!open_set.empty()) {
91  long long current = open_set.top().second;
92  open_set.pop();
93 
94  // Chegou ao destino — reconstrói e retorna o caminho
95  if (current == end_id) {
96  PathResult result;
97  result.node_ids = reconstruct_path(came_from, end_id);
98  result.total_distance = g_score[end_id];
99  result.found = true;
100  return result;
101  }
102 
103  // Explora os vizinhos do nó atual
104  for (const Edge& edge : grafo_.get_neighbors(current)) {
105  double tentative = g_score[current] + edge.distance;
106 
107  // Se encontrou um caminho mais curto até esse vizinho, atualiza
108  if (!g_score.count(edge.toNodeId) || tentative < g_score[edge.toNodeId]) {
109  g_score[edge.toNodeId] = tentative;
110  came_from[edge.toNodeId] = current;
111  double f = tentative + heuristic(edge.toNodeId, end_id);
112  open_set.push({f, edge.toNodeId});
113  }
114  }
115  }
116 
117  // Fila esgotou sem encontrar o destino
118  return {{}, 0.0, false};
119 }
