17 double Grafo::haversine_distance(const MapNode& a, const MapNode& b) const {
18  double distancia, aux, c_;
19  double RAIO = 6371.009; // raio da Terra em km
20 
21  // Converte valores para radianos
22  double phi1 = a.posicao().latitude_ * M_PI / 180.0;
23  double phi2 = b.posicao().latitude_ * M_PI / 180.0;
24  double delta_phi = phi2 - phi1;
25  double delta_lambda = (b.posicao().longitude_ -
26  a.posicao().longitude_) * M_PI / 180.0;
27 
28  // Formula de distância Haversine
29  aux = std::pow(std::sin(delta_phi / 2),2) + std::cos(phi1) *
30  std::cos(phi2) * std::pow(std::sin(delta_lambda / 2),2);
31  c_ = 2 * std::atan2(std::sqrt(aux),std::sqrt(1-aux));
32 
33  // Distancia entre os pontos, em km
34  distancia = RAIO * c_;
35  return distancia;
36 }
