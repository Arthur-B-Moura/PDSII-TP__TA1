199  void parse_tags_lugar(const AttributeList& attributes) {
200  if (attributes.empty()) return;

205  return;
206  }
207 
208  // Atributo "k" indica uma tag OSM chave-valor
209  if (attributes[0].name != "k") return;
210 
211  if (attributes[0].value == "highway") temp_lugar_.tag = "Rua";
212  else if (attributes[0].value == "name") temp_lugar_.nome = attributes[1].value;
213  }
